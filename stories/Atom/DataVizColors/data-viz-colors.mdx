export function readToken(token) {
  const live =
    typeof document !== 'undefined'
      ? getComputedStyle(document.documentElement)
          .getPropertyValue(token)
          .trim()
      : '';
  // Before the partial is imported into the bundle the live value is empty;
  // fall back to the documented literal so the page still renders its numbers.
  return { value: live || FALLBACKS[token] || '', live: Boolean(live) };
}

export const channels = v => v.split(/\s+/).map(Number);
export const toHex = v =>
  '#' +
  channels(v)
    .map(c => c.toString(16).padStart(2, '0'))
    .join('')
    .toUpperCase();

export const srgb = c => {
  const x = c / 255;
  return x > 0.03928 ? Math.pow((x + 0.055) / 1.055, 2.4) : x / 12.92;
};
export const luminance = v => {
  const [r, g, b] = channels(v);
  return 0.2126 * srgb(r) + 0.7152 * srgb(g) + 0.0722 * srgb(b);
};
export const contrast = (a, b) => {
  const [hi, lo] = [luminance(a), luminance(b)].sort((m, n) => n - m);
  return (hi + 0.05) / (lo + 0.05);
};

export function Swatch({ token, note }) {
  const { value, live } = readToken(token);
  if (!value) return null;
  const onWhite = contrast(value, '255 255 255');
  const onBlack = contrast(value, '0 0 0');
  const label = onWhite >= 4.5 ? '255 255 255' : '0 0 0';
  const best = Math.max(onWhite, onBlack);
  return (
    <div
      style={{
        display: 'flex',
        alignItems: 'center',
        gap: '12px',
        marginBottom: '6px',
      }}
    >
      <div
        style={{
          width: '120px',
          height: '44px',
          borderRadius: '4px',
          background: `rgb(${value})`,
          color: `rgb(${label})`,
          border: '1px solid rgba(0,0,0,0.12)',
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          fontSize: '12px',
          flexShrink: 0,
        }}
      >
        {best.toFixed(2)}:1
      </div>
      <div style={{ lineHeight: 1.4 }}>
        <code style={{ fontSize: '13px' }}>{token}</code>
        {!live && (
          <span style={{ fontSize: '11px', color: '#a15' }}> (documented)</span>
        )}
        <div style={{ fontSize: '12px', color: '#555' }}>
          <code>{value}</code> · {toHex(value)} · Y{' '}
          {luminance(value).toFixed(3)} · white {onWhite.toFixed(2)}:1 · black{' '}
          {onBlack.toFixed(2)}:1
        </div>
        {note && <div style={{ fontSize: '12px', color: '#777' }}>{note}</div>}
      </div>
    </div>
  );
}

export function Ramp({ label, tokens, note }) {
  return (
    <div style={{ marginBottom: '28px' }}>
      <h4 style={{ margin: '0 0 8px' }}>{label}</h4>
      {note && (
        <p style={{ margin: '0 0 10px', fontSize: '13px', color: '#555' }}>
          {note}
        </p>
      )}
      <div style={{ display: 'flex', marginBottom: '10px' }}>
        {tokens.map(t => {
          const { value } = readToken(t);
          return (
            <div
              key={t}
              title={`${t} — ${toHex(value)}`}
              style={{
                flex: 1,
                height: '36px',
                background: `rgb(${value})`,
                border: '1px solid rgba(0,0,0,0.08)',
              }}
            />
          );
        })}
      </div>
      {tokens.map(t => (
        <Swatch key={t} token={t} />
      ))}
    </div>
  );
}

# Data visualisation colours

Mangrove had no charting palette. This page documents the one established in
`stories/assets/scss/_tokens-data-viz.scss`. Swatches are read live from the
compiled custom properties; any value marked **(documented)** has not been
wired into the bundle yet and is rendered from the literal recorded in the
source file.

Every contrast ratio on this page is computed in the browser from the resolved
token value using the WCAG 2.2 formula
`(L₁ + 0.05) / (L₂ + 0.05)`, where `L` is relative luminance
`0.2126·R + 0.7152·G + 0.0722·B` over linearised sRGB channels. The figure
inside each swatch is the better of white-on-fill and black-on-fill; the row
beneath gives both.

## Where these colours come from

Nothing here was invented or interpolated.

| Group                     | Source                                                                                                                                         |
| ------------------------- | ---------------------------------------------------------------------------------------------------------------------------------------------- |
| Categorical slots         | Documented Sendai target base colours + the UNDRR brand blue. Only the _ordering_ is a design decision.                                        |
| Sequential blue / magenta | DELTA's shipped chart schemes, `HorizontalBarChart.tsx` → `colorSchemes.cerulean` / `.violet`. Five stops each; they are five stops, not nine. |
| Sequential map            | The Figma "Map Colours" board, "Main maps #296768" stack, all ten swatches.                                                                    |
| Sendai target ramps       | The Figma "Color and Tint Stacks" board, all ten swatches per target.                                                                          |

## Three different jobs

- **Categorical** — unordered series (countries, hazard types, sectors).
- **Sequential** — ordered magnitude (a choropleth, a density scale).
- **Sendai target** — the fixed semantic identity of a Sendai Framework
  target. Damage is Target D whether or not teal suits the page.

Mixing them misleads: a sequential ramp on unordered series implies a
magnitude the data does not have, and categorical hues on a magnitude destroy
the ordering the reader is meant to see.

## Categorical

The slot order is chosen so a short series is colour-blind safe and a long one
degrades predictably. Worst pair for the first _N_ slots, measured as the
smaller of the deuteranope and protanope CIEDE2000 distance
(Viénot–Brettel–Mollon 1999 LMS simulation):

| Series used | Worst pair | Min ΔE₀₀ under CVD | Verdict                      |
| ----------- | ---------- | ------------------ | ---------------------------- |
| 2           | 1 / 2      | 56.8               | safe                         |
| 3           | 1 / 3      | 19.5               | safe                         |
| 4           | 1 / 3      | 19.5               | safe                         |
| 5           | 1 / 5      | 11.1               | marginal                     |
| 6           | 2 / 6      | 8.3                | **not safe on colour alone** |
| 7           | 1 / 7      | 3.2                | **not safe on colour alone** |

**Use at most four slots when colour is the only encoding.** Beyond five, add a
second channel — direct labels, patterns, or shape. Slots 1 and 7 are
effectively the same colour to a protanope (ΔE₀₀ 3.2), and slots 3 and 7
differ by only 6.3 to a deuteranope.

<Ramp label="Categorical slots 1–7" tokens={CATEGORICAL} />

Adjacent-slot contrast ratios are low by design (1.0–2.6:1); series fills are
distinguished by hue and by their own labels, not by luminance against each
other. WCAG has no contrast requirement between two adjacent data marks, but
it does require 4.5:1 for any text drawn on one — which is why the palette
also ships `--mg-dataviz-on-categorical-*`. Only slot 2 (`#D87838`, 3.15:1 on
white) needs black text; every other slot clears 4.5:1 with white.

## Relationship to the existing `--sendai-*` tokens

Mangrove's `--sendai-red|orange|purple|turquoise` ramps are named by colour and
carry no target semantics. Numerically they are a renumbered slice of
Mangrove's own brand palette — `--sendai-red-400` is byte-identical to
`--mg-color-red-600`, `--sendai-red-900` to `--mg-color-red-900`, and so on —
so they encode PreventionWeb brand hues, not Sendai Framework identities.
They do not agree with the target ramps above:

| Existing token                     | Nearest target ramp | ΔE₀₀ | Same colour?          |
| ---------------------------------- | ------------------- | ---- | --------------------- |
| `--sendai-purple-900` `#962987`    | Target B `#984089`  | 4.5  | no                    |
| `--sendai-orange-900` `#eb752a`    | Target C `#D87838`  | 3.7  | no                    |
| `--sendai-turquoise-900` `#00afae` | Target D `#037F7F`  | 16.0 | no                    |
| `--sendai-red-900` `#c10920`       | Target G `#833640`  | 15.3 | no                    |
| —                                  | Targets A, E, F     | —    | no counterpart at all |

The new tokens are additive. Nothing in this file changes `--sendai-*`; the two
sets are documented side by side so the conflict is visible rather than
averaged away.

## Chart chrome

`--mg-dataviz-surface`, `--mg-dataviz-gridline`, `--mg-dataviz-axis`,
`--mg-dataviz-label` and `--mg-dataviz-muted-label` alias the neutral palette
so a theme swap carries them. Gridlines sit below 3:1 on purpose — they are
decoration and SC 1.4.11 does not apply. The axis line (3.96:1) and tick
labels (12.63:1 / 5.74:1 on white) are informational and do meet it.
